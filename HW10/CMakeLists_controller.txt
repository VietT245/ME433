cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)

project(mpu6050_controller C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

add_executable(mpu6050_controller
    mpu6050_controller.c
)

pico_enable_stdio_usb(mpu6050_controller 1)
pico_enable_stdio_uart(mpu6050_controller 0)

target_link_libraries(mpu6050_controller
    pico_stdlib
    hardware_i2c
    hardware_gpio
)

pico_add_extra_outputs(mpu6050_controller)
